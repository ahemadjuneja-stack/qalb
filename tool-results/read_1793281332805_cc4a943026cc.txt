    53→    53→function withTimeout<T>(p: Promise<T>, ms: number): Promise<T> {
    54→    54→  return new Promise<T>((resolve, reject) => {
    55→    55→    const timer = setTimeout(() => reject(new Error('search-timeout')), ms)
    56→    56→    p.then(
    57→    57→      (v) => { clearTimeout(timer); resolve(v) },
    58→    58→      (e) => { clearTimeout(timer); reject(e) }
    59→    59→    )
    60→    60→  })
    61→    61→}

   118→   118→  async function runSearch(rawQuery: string) {

   157→   157→          title: 'Koi paak kalaam result nahi mila',

   166→   166→        title: 'Search me masla hua',
