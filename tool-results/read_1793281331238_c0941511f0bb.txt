    51→const SEARCH_WATCHDOG_MS = 15000
    52→
    53→function withTimeout<T>(p: Promise<T>, ms: number): Promise<T> {
    54→  return new Promise<T>((resolve, reject) => {
    55→    const timer = setTimeout(() => reject(new Error('search-timeout')), ms)
    56→    p.then(
    57→      (v) => { clearTimeout(timer); resolve(v) },
    58→      (e) => { clearTimeout(timer); reject(e) }
    59→    )
    60→  })
    61→}

   118→  async function runSearch(rawQuery: string) {
   119→    const q = rawQuery.trim()
   120→    if (!q) return
   121→    // NOTE: pehle yahan phase-guard tha — agar koi request atak jati to
   122→    // search HAMESHA ke liye lock ho jata tha. Ab nayi search purani ko
   123→    // cancel/supersede karti hai (stale generation ka result discard).
   124→    const searchId = ++searchIdRef.current
   125→    const stale = () => searchIdRef.current !== searchId
   126→    setQuery(q)
   127→    setGuardian(null)
   128→    setPhase('validating')
   129→    try {
   130→      // APK-safe chain: hosted API → device-side rules (search-client.ts)
   131→      const validation = await withTimeout(validateSmart(q), SEARCH_WATCHDOG_MS)
   132→      if (stale()) return
   133→      setGuardian(validation)
   134→
   135→      if (!validation.valid) {
   136→        setPhase('blocked')
   137→        return
   138→      }
   139→
   140→      setPhase('searching')
   141→      // APK-safe chain: YouTube unlimited → YT API → Firestore → bundled
   142→      const usedQuery = validation.refinedQuery || q
   143→      setRefined(usedQuery)
   144→      // strict mode: ambiguous query (naam waghera) → results par sirf
   145→      // Islamic-marker wale pass hote hain (Bollywood/bhajan hide)
   146→      const strict = validation.strict === true
   147→      setStrictMode(strict)
   148→      const data = await withTimeout(youtubeSearchSmart(usedQuery, null, strict), SEARCH_WATCHDOG_MS)
   149→      if (stale()) return
   150→      setResults(data.tracks)
   151→      setResultSource(data.source)
   152→      setActiveContinuation(data.continuation)
   153→      setPhase('results')
   154→
   155→      if (data.tracks.length === 0) {
   156→        toast({
   157→          title: 'Koi paak kalaam result nahi mila',
   158→          description:
   159→            'Islamic keywords ke saath search karein — jaise "Tajdar-e-Haram naat" ya "Nusrat qawwali".',
   160→        })
   161→      }
   162→    } catch {
   163→      if (stale()) return
   164→      setPhase('idle')
   165→      toast({
   166→        title: 'Search me masla hua',
   167→        description: 'Network slow hai — dobara try karein.',
   168→      })
   169→    }
   170→  }

   320→      {/* blocked state */}
